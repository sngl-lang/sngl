# Canvas2D Component Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement declarative Canvas2D drawing: `canvas` + shape components in SNGL, `passCanvas` lower pass that transforms shape children into a draw function, and HTML5 canvas rendering.

**Architecture:** Shape components carry `list<shape>` ChildrenType. The `passCanvas` lower pass walks the IR, collects shape children from canvas NodeInsts, and generates a draw function (ir.Func with canvas intrinsic calls). The HTML platform emits `<canvas>` elements and translates the draw function to JS Canvas2D API calls. `#[canvas.shape]` macro rewrites user component ChildrenType to `list<shape>`. Stdlib shape components declare `list<shape>` directly.

**Tech Stack:** Go, `ir/`, `internal/lower/`, `codegen/platform/html/`, existing `codegen.RegisterIntrinsic` pattern.

```go
func shapeHandler(attr ast.MacroAttr, decl ast.Stmt) (ast.Stmt, error) {
    comp, ok := decl.(*ast.ComponentDecl)
    if !ok {
        return decl, errors.New("shape macro requires a component declaration")
    }
    for _, p := range comp.Props.Props {
        if _, isEvent := p.(ast.EventDecl); isEvent {
            return decl, errors.New("shape components do not support event declarations")
        }
    }

    // Rewrite ChildrenType to list<shape>.
    listShape := &ast.NamedType{
        Name:     "list",
        TypeArgs: []ast.TypeExpr{&ast.NamedType{Name: "shape"}},
    }
    switch {
    case comp.ChildrenType == nil:
        comp.ChildrenType = listShape
    default:
        named, isNamed := comp.ChildrenType.(*ast.NamedType)
        if isNamed && named.Name == "list" && len(named.TypeArgs) > 0 {
            inner, ok := named.TypeArgs[0].(*ast.NamedType)
            if ok && inner.Name == "shape" {
                // already list<shape> — leave unchanged
            } else if ok && inner.Name == "component" {
                comp.ChildrenType = listShape
            } else {
                return decl, errors.New("shape components may only have list<shape> children")
            }
        } else {
            return decl, errors.New("shape components may only have list<shape> children")
        }
    }

    comp.IsShape = true
    return comp, nil
}
```

```sngl
component circle(cx Length, cy Length, r Length, style CanvasStyle) list<shape> {}

component ellipse(cx Length, cy Length, rx Length, ry Length, style CanvasStyle) list<shape> {}
```

Note: ALL stdlib shape components declare `list<shape>` ChildrenType directly — they don't use `#[canvas.shape]`. The macro is for user-defined shape components. `list<shape>` is also what makes a component detectable as a "shape" by the checker's child validation (Task 2 checks `ChildrenType.Elems[0].Kind == TypeShape`). Leaf shapes like `line`, `text`, `path`, `image` accept `list<shape>` children syntactically; the lower pass ignores those nested children since there is no draw primitive for nesting on leaf shapes.

```go
func TestPassCanvas_GeneratesDrawFunc(t *testing.T) {

    caps := lower.Caps{Canvas: true}
    if err := lower.Lower(pkg, caps, lower.Options{}); err != nil {
        t.Fatalf("lower error: %v", err)

    if canvasInst.CanvasDraw == nil {
```

```go
package lower

import (
    "fmt"

    "git.duckfam.us/jonathan/sngl/ir"
)

var passCanvas = pass{
    name:    "Canvas",
    enabled: func(c Caps) bool { return c.Canvas },
    apply:   lowerCanvas,
}

func lowerCanvas(pkg *ir.Package, _ Caps, _ Options) error {
    if pkg == nil {
        return nil
    }
    for _, comp := range pkg.Components {
        comp.Body = processCanvasStmts(comp.Body, &comp.Funcs)
    }
    for _, w := range pkg.Windows {
        w.Body = processCanvasStmts(w.Body, &w.Funcs)
    }
    return nil
}

// processCanvasStmts walks statements and transforms canvas NodeInsts.
func processCanvasStmts(stmts []ir.Stmt, funcs *[]*ir.Func) []ir.Stmt {
    for _, s := range stmts {
        ni, ok := s.(*ir.NodeInst)
        if !ok {
            continue
        }
        if !isShapeContainer(ni) {
            continue
        }
        drawFunc := generateDrawFunc(ni.Children, funcs)
        *funcs = append(*funcs, drawFunc)
        ni.CanvasDraw = drawFunc
        ni.Children = nil
    }
    return stmts
}

// isShapeContainer reports whether a NodeInst has list<shape> children type.
func isShapeContainer(ni *ir.NodeInst) bool {
    if ni.Component == nil || ni.Component.ChildrenType == nil {
        return false
    }
    ct := ni.Component.ChildrenType
    return ct.Kind == ir.TypeList &&
        len(ct.Elems) > 0 &&
        ct.Elems[0].Kind == ir.TypeShape
}

// canvasNS returns the canvas namespace ident used as receiver for intrinsic calls.
func canvasNS() *ir.Ident {
    return &ir.Ident{Name: "canvas", Type: ir.TypDyn}
}

// ctxIdent returns an Ident for the draw function's ctx parameter.
func ctxIdent() *ir.Ident {
    return &ir.Ident{Name: "ctx", Type: ir.TypDyn}
}

// canvasIntrinsicFunc materializes a CanvasIntrinsic into *ir.Func form.
func canvasIntrinsicFunc(name string) *ir.Func {
    def := ir.LookupIntrinsic(name)
    if def == nil {
        panic(fmt.Sprintf("passCanvas: unknown intrinsic %q", name))
    }
    return &ir.Func{
        Name:      def.Name,
        Intrinsic: def.Name,
        Params:    def.Params,
        Return:    def.Return,
    }
}

// callCanvas emits a CallStmt calling a canvas intrinsic with ctx + extra args.
func callCanvas(intrinsicName string, args ...ir.Expr) *ir.CallStmt {
    allArgs := make([]ir.CallArg, 0, 1+len(args))
    allArgs = append(allArgs, ir.CallArg{Name: "ctx", Value: ctxIdent()})
    for _, a := range args {
        allArgs = append(allArgs, ir.CallArg{Value: a})
    }
    return &ir.CallStmt{
        Call: &ir.Call{
            Type:     ir.TypVoid,
            Receiver: canvasNS(),
            Func:     canvasIntrinsicFunc(intrinsicName),
            Args:     allArgs,
        },
    }
}

// findArg returns the value of a named prop arg on a NodeInst, or nil.
func findArg(ni *ir.NodeInst, name string) ir.Expr {
    for i := range ni.Props {
        if ni.Props[i].Name == name {
            return ni.Props[i].Value
        }
    }
    return nil
}

var zero ir.Expr = &ir.NumberLiteral{Value: 0}

// generateDrawFunc builds a draw function for a canvas container's shape children.
// funcs is the parent component's Funcs slice (used for recursive canvas groups).
func generateDrawFunc(children []ir.Stmt, funcs *[]*ir.Func) *ir.Func {
    var body []ir.Stmt
    emitShapes(children, &body, funcs)

    return &ir.Func{
        Name:   "_canvasDraw",
        Params: []*ir.Param{{Name: "ctx", Type: ir.TypDyn}},
        Return: ir.TypVoid,
        Body:   body,
    }
}

// emitShapes recursively emits draw calls for a list of shape children.
func emitShapes(children []ir.Stmt, body *[]ir.Stmt, funcs *[]*ir.Func) {
    for _, s := range children {
        ni, ok := s.(*ir.NodeInst)
        if !ok {
            continue
        }
        emitShape(ni, body, funcs)
    }
}

// emitShape emits save/applyStyle/drawPrimitive/[children]/restore for one shape.
func emitShape(ni *ir.NodeInst, body *[]ir.Stmt, funcs *[]*ir.Func) {
    *body = append(*body, callCanvas("CanvasSave"))

    // applyStyle if a style prop exists.
    if styleExpr := findArg(ni, "style"); styleExpr != nil {
        *body = append(*body, callCanvas("CanvasApplyStyle", styleExpr))
    }

    // Emit the primitive draw call based on component name.
    switch ni.Name {
    case "rect":
        x := findArgOrZero(ni, "x")
        y := findArgOrZero(ni, "y")
        w := findArgOrZero(ni, "w")
        h := findArgOrZero(ni, "h")
        *body = append(*body, callCanvas("CanvasDrawRect", x, y, w, h))
    case "circle":
        cx := findArgOrZero(ni, "cx")
        cy := findArgOrZero(ni, "cy")
        r := findArgOrZero(ni, "r")
        *body = append(*body, callCanvas("CanvasDrawCircle", cx, cy, r))
    case "ellipse":
        cx := findArgOrZero(ni, "cx")
        cy := findArgOrZero(ni, "cy")
        rx := findArgOrZero(ni, "rx")
        ry := findArgOrZero(ni, "ry")
        *body = append(*body, callCanvas("CanvasDrawEllipse", cx, cy, rx, ry))
    case "line":
        x1 := findArgOrZero(ni, "x1")
        y1 := findArgOrZero(ni, "y1")
        x2 := findArgOrZero(ni, "x2")
        y2 := findArgOrZero(ni, "y2")
        *body = append(*body, callCanvas("CanvasDrawLine", x1, y1, x2, y2))
    case "path":
        cmds := findArgOrZero(ni, "cmds")
        *body = append(*body, callCanvas("CanvasDrawPath", cmds))
    case "text":
        x := findArgOrZero(ni, "x")
        y := findArgOrZero(ni, "y")
        content := findArgOrZero(ni, "content")
        *body = append(*body, callCanvas("CanvasDrawText", x, y, content))
    case "image":
        x := findArgOrZero(ni, "x")
        y := findArgOrZero(ni, "y")
        w := findArgOrZero(ni, "w")
        h := findArgOrZero(ni, "h")
        src := findArgOrZero(ni, "src")
        *body = append(*body, callCanvas("CanvasDrawImage", x, y, w, h, src))
    // User-defined shapes (unknown names): only emit children as a group.
    }

    // Recurse into nested shape children.
    if len(ni.Children) > 0 {
        emitShapes(ni.Children, body, funcs)
    }

    *body = append(*body, callCanvas("CanvasRestore"))
}

func findArgOrZero(ni *ir.NodeInst, name string) ir.Expr {
    if v := findArg(ni, name); v != nil {
        return v
    }
    return &ir.NumberLiteral{Value: 0}
}
```

# C APIs for SNGL Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `c://` import scheme that parses C headers at compile time and exposes their declarations as SNGL-typed functions, structs, and enums, with `golang`/cgo as the first language translator implementation.

**Architecture:** `codegen/scheme/c` registers a `SchemeImporter` using `modernc.org/cc/v4` to parse headers; `CCompiler` optional interface on `LangTranslator` (same pattern as `WASMCompiler`) lets the `golang` translator emit the cgo preamble + `import "C"`. The fyne platform wires these together in its codegen.

**Tech Stack:** Go, `modernc.org/cc/v4` v4.28.2, cgo, `pkg-config` (optional for `c://pkg:` URIs)

```go
package c

import (
	"git.duckfam.us/jonathan/sngl/ir"
	"modernc.org/cc/v4"
)

// mapCType converts a cc.Type to an *ir.Type.
// structs is a mutable map from C struct name → ir.StructDef; entries are
// added on first encounter so recursive/shared struct pointers reuse the same def.
// Returns nil for unmappable types (caller should set Func.Unusable).
func mapCType(t cc.Type, ast *cc.AST, structs map[string]*ir.StructDef) *ir.Type {
	if t == nil {
		return nil
	}
	switch t.Kind() {
	case cc.Bool:
		return &ir.Type{Kind: ir.TypeBool}
	case cc.Char, cc.SChar, cc.UChar:
		return &ir.Type{Kind: ir.TypeInt}
	case cc.Int, cc.Short, cc.Long, cc.LongLong,
		cc.UInt, cc.UShort, cc.ULong, cc.ULongLong:
		return &ir.Type{Kind: ir.TypeInt}
	case cc.Float, cc.Double, cc.LongDouble:
		return &ir.Type{Kind: ir.TypeFloat}
	case cc.Void:
		return nil // void return
	case cc.Ptr:
		return mapPointerType(t, ast, structs)
	case cc.Struct:
		return mapStructType(t, structs)
	case cc.Enum:
		return mapEnumType(t)
	case cc.Array:
		// T[] → []T
		elem := mapCType(t.Elem(), ast, structs)
		if elem == nil {
			return nil
		}
		return &ir.Type{Kind: ir.TypeList, Elems: []*ir.Type{elem}}
	case cc.Function, cc.Union:
		return nil // unmappable
	default:
		return nil
	}
}

func mapPointerType(t cc.Type, ast *cc.AST, structs map[string]*ir.StructDef) *ir.Type {
	elem := t.Elem()
	if elem == nil {
		return &ir.Type{Kind: ir.TypeDyn, Meta: "unsafe.Pointer"}
	}
	switch elem.Kind() {
	case cc.Void:
		// void* → unsafe.Pointer
		return &ir.Type{Kind: ir.TypeDyn, Meta: "unsafe.Pointer"}
	case cc.Char, cc.SChar:
		// char* → string (cgo handles the C.GoString / C.CString conversion)
		return &ir.Type{Kind: ir.TypeString}
	case cc.Struct:
		inner := mapStructType(elem, structs)
		if inner == nil {
			return &ir.Type{Kind: ir.TypeDyn, Meta: "unsafe.Pointer"}
		}
		// T* → optional<ref<T>>
		return &ir.Type{Kind: ir.TypeOption, Elems: []*ir.Type{
			{Kind: ir.TypeRef, Elems: []*ir.Type{inner}},
		}}
	case cc.Union:
		// union* → unsafe.Pointer (unusable in SNGL)
		return &ir.Type{Kind: ir.TypeDyn, Meta: "unsafe.Pointer"}
	default:
		inner := mapCType(elem, ast, structs)
		if inner == nil {
			return &ir.Type{Kind: ir.TypeDyn, Meta: "unsafe.Pointer"}
		}
		return &ir.Type{Kind: ir.TypeOption, Elems: []*ir.Type{
			{Kind: ir.TypeRef, Elems: []*ir.Type{inner}},
		}}
	}
}

func mapStructType(t cc.Type, structs map[string]*ir.StructDef) *ir.Type {
	st, ok := t.(*cc.StructType)
	if !ok {
		return nil
	}
	name := st.Tag().SrcStr() // cc.Token.SrcStr() returns the identifier string
	if name == "" {
		return nil // anonymous struct → skip
	}
	if structs == nil {
		structs = map[string]*ir.StructDef{}
	}
	if sd, ok := structs[name]; ok {
		return &ir.Type{Kind: ir.TypeStruct, Decl: sd}
	}
	// Create struct def with Native set to "C.Name" so IRTypeToGo emits the cgo type.
	sd := &ir.StructDef{
		Name:   name,
		Native: "C." + name,
	}
	structs[name] = sd
	// Map fields (best-effort; unmappable fields are skipped).
	for i := 0; i < st.NumFields(); i++ {
		f := st.FieldByIndex(i)
		if f.Name() == "" || f.IsBitfield() { // IsBitfield — lowercase 'f'
			continue
		}
		ft := mapCType(f.Type(), nil, structs)
		if ft == nil {
			continue
		}
		sd.Fields = append(sd.Fields, &ir.StructField{ // ir.StructField, not ir.FieldDef
			Name:       f.Name(),
			NativeName: f.Name(),
			Type:       ft,
		})
	}
	return &ir.Type{Kind: ir.TypeStruct, Decl: sd}
}

func mapEnumType(t cc.Type) *ir.Type {
	et, ok := t.(*cc.EnumType)
	if !ok {
		return &ir.Type{Kind: ir.TypeInt}
	}
	name := et.Tag().SrcStr() // cc.Token.SrcStr() returns the identifier string
	ed := &ir.EnumDef{Name: name}
	for _, e := range et.Enumerators() { // Enumerators() returns []*cc.Enumerator; no index API
		ed.Members = append(ed.Members, &ir.EnumMember{ // ir.EnumMember, not ir.EnumValue
			Name: e.Token.SrcStr(), // enumerator name from Token.SrcStr()
		})
	}
	return &ir.Type{Kind: ir.TypeEnum, Decl: ed}
}

package c

import (
	"fmt"
	"os/exec"
	"runtime"
	"strings"

	"git.duckfam.us/jonathan/sngl/codegen"
	"git.duckfam.us/jonathan/sngl/ir"
	"modernc.org/cc/v4"
)

// predefinedPreamble is the minimal predefined C environment needed by cc/v4.
const predefinedPreamble = `
int __predefined_declarator;
#if defined(__i386__) || defined(__arm__)
typedef unsigned __predefined_size_t;
#else
typedef unsigned long long __predefined_size_t;
#endif
`

func init() {
	codegen.RegisterScheme(&CImporter{})
}

// CImporter resolves c:// scheme imports by parsing C headers.
type CImporter struct{}

func (c *CImporter) Scheme() string { return "c" }

// Resolve handles two URI forms:
//   - c://path/to/header.h — absolute or dir-relative header file
//   - c://pkg:name         — pkg-config managed library
func (c *CImporter) Resolve(uri, dir string) (*ir.NativeImport, error) {
	path := strings.TrimPrefix(uri, "c://")
	if path == "" {
		return nil, fmt.Errorf("c scheme requires a path (e.g. c:///usr/include/SDL2/SDL.h or c://pkg:sdl2)")
	}

	if strings.HasPrefix(path, "pkg:") {
		return c.resolvePkgConfig(strings.TrimPrefix(path, "pkg:"), dir)
	}
	return c.resolveHeader(path, nil, uri)
}

// resolvePkgConfig uses pkg-config to find include paths and the primary header.
func (c *CImporter) resolvePkgConfig(libName, dir string) (*ir.NativeImport, error) {
	// Verify pkg-config is available.
	if _, err := exec.LookPath("pkg-config"); err != nil {
		return nil, fmt.Errorf("pkg-config required for c://pkg: imports — install it or use a direct header path")
	}

	// Get include paths.
	cflagsOut, err := exec.Command("pkg-config", "--cflags", libName).Output()
	if err != nil {
		return nil, fmt.Errorf("pkg-config --cflags %s: %w", libName, err)
	}
	includePaths := parsePkgConfigCflags(string(cflagsOut))

	// Get link flags.
	libsOut, err := exec.Command("pkg-config", "--libs", libName).Output()
	if err != nil {
		return nil, fmt.Errorf("pkg-config --libs %s: %w", libName, err)
	}
	linkFlags := strings.Fields(strings.TrimSpace(string(libsOut)))

	// Find the primary header: conventionally <libName>/<libName>.h or <libName>.h.
	// pkg-config doesn't specify a canonical header, so we use the first include path.
	if len(includePaths) == 0 {
		return nil, fmt.Errorf("pkg-config --cflags %s returned no -I paths", libName)
	}

	// Build a synthetic header that includes the library's top-level header.
	// Convention: if libName is "gtk+-3.0" → try "gtk/gtk.h"; "sdl2" → "SDL2/SDL.h".
	syntheticInclude := inferPkgHeader(libName)
	syntheticSrc := fmt.Sprintf("#include <%s>\n", syntheticInclude)

	ni, err := c.resolveHeaderWithIncludes(syntheticSrc, includePaths, "c://pkg:"+libName)
	if err != nil {
		return nil, err
	}
	ni.LinkFlags = linkFlags
	return ni, nil
}

// resolveHeader parses a single header file at path.
func (c *CImporter) resolveHeader(path string, includePaths []string, importPath string) (*ir.NativeImport, error) {
	src := fmt.Sprintf("#include %q\n", path)
	return c.resolveHeaderWithIncludes(src, includePaths, importPath)
}

// resolveHeaderWithIncludes runs cc.Translate on src with the given include paths.
func (c *CImporter) resolveHeaderWithIncludes(src string, includePaths []string, importPath string) (*ir.NativeImport, error) {
	abi, err := cc.NewABI(runtime.GOOS, runtime.GOARCH)
	if err != nil {
		return nil, fmt.Errorf("cc.NewABI: %w", err)
	}
	cfg := &cc.Config{
		ABI:          abi,
		IncludePaths: includePaths,
	}
	ast, err := cc.Translate(cfg, []cc.Source{
		{Name: "<predefined>", Value: predefinedPreamble},
		{Name: "<import>", Value: src},
	})
	if err != nil {
		return nil, fmt.Errorf("parsing C header: %w", err)
	}

	return extractDeclarations(ast, importPath)
}

// extractDeclarations walks the cc AST scope and produces an ir.NativeImport.
func extractDeclarations(ast *cc.AST, importPath string) (*ir.NativeImport, error) {
	ni := &ir.NativeImport{ImportPath: importPath}
	structs := map[string]*ir.StructDef{}

	for _, nodes := range ast.Scope.Nodes {
		for _, node := range nodes {
			d, ok := node.(*cc.Declarator)
			if !ok || d.IsSynthetic() {
				continue
			}
			name := d.Name()
			if name == "" || strings.HasPrefix(name, "__") {
				continue
			}
			t := d.Type()
			if t == nil {
				continue
			}

			switch t.Kind() {
			case cc.Function:
				fn := extractFunc(d, name, t.(*cc.FunctionType), ast, structs)
				if fn != nil {
					ni.Funcs = append(ni.Funcs, fn)
				}
			case cc.Struct:
				sd := extractStruct(t, name, structs)
				if sd != nil {
					ni.Structs = appendIfNew(ni.Structs, sd)
				}
			case cc.Enum:
				ed := extractEnum(t, name)
				if ed != nil {
					ni.Enums = append(ni.Enums, ed)
				}
			}
		}
	}

	// Collect any structs referenced by functions that weren't directly named.
	for _, sd := range structs {
		ni.Structs = appendIfNew(ni.Structs, sd)
	}

	return ni, nil
}

func extractFunc(d *cc.Declarator, name string, ft *cc.FunctionType, ast *cc.AST, structs map[string]*ir.StructDef) *ir.Func {
	fn := &ir.Func{
		Name:       name,
		NativePkg:  "C",
		NativeName: "C." + name,
	}

	// Return type.
	ret := mapCType(ft.Result(), ast, structs)
	if ft.Result() != nil && ft.Result().Kind() != cc.Void {
		if ret == nil {
			fn.Unusable = fmt.Sprintf("C function %s has unmappable return type", name)
			return fn
		}
		fn.Return = ret
	}

	// Parameters.
	for _, p := range ft.Parameters() {
		pt := mapCType(p.Type(), ast, structs)
		if pt == nil {
			fn.Unusable = fmt.Sprintf("C function %s param %s has unmappable type", name, p.Name())
			return fn
		}
		pname := p.Name()
		if pname == "" {
			pname = fmt.Sprintf("arg%d", len(fn.Params))
		}
		fn.Params = append(fn.Params, &ir.Param{Name: pname, Type: pt})
	}

	return fn
}

func extractStruct(t cc.Type, name string, structs map[string]*ir.StructDef) *ir.StructDef {
	mapStructType(t, structs)
	return structs[name]
}

func extractEnum(t cc.Type, name string) *ir.EnumDef {
	irType := mapEnumType(t)
	if irType == nil {
		return nil
	}
	ed, ok := irType.Decl.(*ir.EnumDef)
	if !ok {
		return nil
	}
	if ed.Name == "" {
		ed.Name = name // fall back to declarator name for anonymous enums
	}
	return ed
}

func appendIfNew(slice []*ir.StructDef, sd *ir.StructDef) []*ir.StructDef {
	for _, existing := range slice {
		if existing.Name == sd.Name {
			return slice
		}
	}
	return append(slice, sd)
}

func parsePkgConfigCflags(cflags string) []string {
	var paths []string
	for _, field := range strings.Fields(cflags) {
		if strings.HasPrefix(field, "-I") {
			paths = append(paths, strings.TrimPrefix(field, "-I"))
		}
	}
	return paths
}

// inferPkgHeader guesses the primary header for a pkg-config library name.
// Examples: "gtk+-3.0" → "gtk/gtk.h", "sdl2" → "SDL2/SDL.h".
func inferPkgHeader(libName string) string {
	// Normalize: strip version suffix (e.g. "gtk+-3.0" → "gtk+")
	base := libName
	if idx := strings.LastIndex(base, "-"); idx != -1 {
		base = base[:idx]
	}
	base = strings.TrimRight(base, "+")
	base = strings.ToUpper(base[:1]) + base[1:]
	// Best-effort: try <Base/Base.h>
	return base + "/" + base + ".h"
}

package golang

import (
	"strings"

	"git.duckfam.us/jonathan/sngl/ir"
)

// EmitCHeader implements codegen.CCompiler for the Go/cgo translator.
// It emits the cgo comment preamble and `import "C"` line required before
// other imports in any Go file that calls C functions.
//
// Returns empty string when imports is nil or empty.
func (t *Translator) EmitCHeader(imports []*ir.NativeImport) string {
	if len(imports) == 0 {
		return ""
	}

	var b strings.Builder
	b.WriteString("/*\n")

	// Collect link flags from all imports.
	seen := map[string]bool{}
	for _, ni := range imports {
		for _, flag := range ni.LinkFlags {
			if !seen[flag] {
				seen[flag] = true
			}
		}
	}
	if len(seen) > 0 {
		// Separate -l flags (LDFLAGS) from -L flags (LDFLAGS) and -I flags (CFLAGS).
		var ldflags, cflags []string
		for flag := range seen {
			if strings.HasPrefix(flag, "-I") {
				cflags = append(cflags, flag)
			} else {
				ldflags = append(ldflags, flag)
			}
		}
		if len(cflags) > 0 {
			b.WriteString("#cgo CFLAGS: " + strings.Join(cflags, " ") + "\n")
		}
		if len(ldflags) > 0 {
			b.WriteString("#cgo LDFLAGS: " + strings.Join(ldflags, " ") + "\n")
		}
	}

	// Emit #include for each import path.
	for _, ni := range imports {
		headerPath := strings.TrimPrefix(ni.ImportPath, "c://")
		if strings.HasPrefix(headerPath, "pkg:") {
			// pkg: imports use angle-bracket includes with inferred header.
			// The synthetic include was already embedded in the parsing step;
			// here we emit the canonical include line.
			libName := strings.TrimPrefix(headerPath, "pkg:")
			b.WriteString("#include <" + inferPkgHeaderForEmit(libName) + ">\n")
		} else {
			b.WriteString("#include \"" + headerPath + "\"\n")
		}
	}

	b.WriteString("*/\nimport \"C\"\n")
	return b.String()
}

// inferPkgHeaderForEmit mirrors inferPkgHeader from the c scheme importer.
// Kept here to avoid an import cycle (golang → scheme/c would be circular).
func inferPkgHeaderForEmit(libName string) string {
	base := libName
	if idx := strings.LastIndex(base, "-"); idx != -1 {
		base = base[:idx]
	}
	base = strings.TrimRight(base, "+")
	if len(base) == 0 {
		return libName + ".h"
	}
	base = strings.ToUpper(base[:1]) + base[1:]
	return base + "/" + base + ".h"
}
```

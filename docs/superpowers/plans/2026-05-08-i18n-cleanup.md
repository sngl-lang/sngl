```markdown
### Runtime packages for generated code

SNGL ships per-target-language runtime packages under `pkg/<lang>/<name>/`. These contain Go/Kotlin/JS code that generated programs import. Examples:

- `pkg/go/i18n/` — Go runtime backing the `i18n` SNGL stdlib package (manifest loader, ICU formatter, date/number formatters wrapping `golang.org/x/text` and `github.com/goodsign/monday`).

The `lib/` directory holds **SNGL stdlib declarations** (language-agnostic `.sngl` files embedded into the compiler). The `pkg/` directory holds **runtime implementations** (per-target-language packages emitted into generated code's import graph).

When adding a new stdlib package that needs runtime support:
1. Declare the SNGL surface in `lib/<name>.sngl`.
2. For each target language that needs runtime support, create `pkg/<lang>/<name>/`.
3. The codegen for that language emits `import "git.duckfam.us/jonathan/sngl/pkg/<lang>/<name>"` and translates stdlib calls to that package's API.
```

```go
// checkExprExpecting type-checks expr in a context expecting `want`.
// For *ast.MapLit and *ast.StructExpr, this provides the expected-type

func (c *checker) checkExprExpecting(expr ast.Expr, want *ir.Type) ir.Expr {
	if want == nil {
		return c.checkExpr(expr)
```

- [ ] **Step 1: Update the production**

Find the existing `I18nPlaceholder` production in `sngl.ebnf`:

```
I18nPlaceholder = Expr [ comma ident [ comma MsgFormatBody ] ] .
```

Change the third-arg alternative to allow either a bare ident or a MsgFormatBody:

```
I18nPlaceholder = Expr [ comma ident [ comma I18nThirdArg ] ] .
I18nThirdArg    = ident | MsgFormatBody .
```

The egg generator may need this expressed differently if there's an LL(1) conflict. The first set of `MsgFormatBody` is `Selector` which is `ident | assign int_lit`. So `ident` could start either alternative. To disambiguate: if the ident is followed by `{`, it's a MsgFormatBody (Selector → ident → `{` body); otherwise it's a bare style ident (no `{` follows).

If egg can't handle that lookahead, refactor:

```
I18nThirdArg    = ident [ MsgBody { MsgCase } ] | "=" int_lit MsgBody { MsgCase } .
```

(The bracket means the body is optional. If present, the leading ident is a Selector; if absent, it's a Style.)

```go
func (b *builder) buildI18nPlaceholder(it nodeIter) ast.Expr {
	ph := &ast.I18nPlaceholderExpr{}

		ph.Value = b.buildExpr(it.enter())
		if ph.Value != nil {
```

### Task 10: ICU `'...'` apostrophe quoting

**Files:**
- Modify: `internal/parser/lexer.go`
- Test: `testdata/i18n_apostrophe.sngl` (existing)

ICU MessageFormat treats `'X'` as a quoted run of literal characters (for X containing `{`, `}`, `#`, or other metachars). `''` (two apostrophes) emits a single literal apostrophe. The lexer needs to handle this inside i18n string content.

- [ ] **Step 1: Find `scanStringContent`**

```bash

```

- [ ] **Step 2: Add apostrophe handling inside i18n mode**

Inside `scanStringContent`'s main scan loop, when `i18n == true` and the current char is `'`:

```go
if i18n && ch == '\'' {
	// Doubled '' → emit single literal '.
	if l.pos+1 < len(l.input) && l.input[l.pos+1] == '\'' {
		sb.WriteByte('\'')
		l.advance()
		l.advance()
		continue
	}
	// 'text' quoted run — the next char must be a metachar to start a quoted run;
	// otherwise the apostrophe is literal (matches ICU's behavior).
	next := byte(0)
	if l.pos+1 < len(l.input) {
		next = byte(l.input[l.pos+1])
	}
	if next == '{' || next == '}' || next == '#' || next == '|' {
		// Quoted run: copy verbatim until matching closing apostrophe.
		l.advance() // consume opening '
		for l.pos < len(l.input) {
			if l.input[l.pos] == '\'' {
				// '' inside a quoted run still means a literal '.
				if l.pos+1 < len(l.input) && l.input[l.pos+1] == '\'' {
					sb.WriteByte('\'')
					l.advance()
					l.advance()
					continue
				}
				l.advance() // consume closing '
				break
			}
			sb.WriteRune(l.input[l.pos])
			l.advance()
		}
		continue
	}
	// Bare ' not preceded/followed by a metachar — literal apostrophe.
	sb.WriteByte('\'')
	l.advance()
	continue
}
```

This is ICU's actual rule: `'` only starts a quoted run if the next char is a metachar; otherwise it's literal.

```go
// Date formats a time.Time as a date for the translator's locale.
// Style: "short", "medium" (default), "long", "full".
func (t *Translator) Date(d time.Time, style string) string {
	return monday.Format(d, dateLayout(style), localeFor(t.Locale))
}

func (t *Translator) Time(d time.Time, style string) string {
	return monday.Format(d, timeLayout(style), localeFor(t.Locale))
}

func (t *Translator) Datetime(d time.Time, dateStyle, timeStyle string) string {
	layout := dateLayout(dateStyle) + " " + timeLayout(timeStyle)
	return monday.Format(d, layout, localeFor(t.Locale))
}

// dateLayout maps an ICU style name to a Go time-layout string. Locale
// substitutions for month/day names are handled by monday.Format.
func dateLayout(style string) string {
	switch style {
	case "short":
		return "1/2/06"
	case "long":
		return "January 2, 2006"
	case "full":
		return "Monday, January 2, 2006"
	}
	return "Jan 2, 2006" // medium
}

func timeLayout(style string) string {
	switch style {
	case "short":
		return "3:04 PM"
	case "long", "full":
		return "3:04:05 PM MST"
	}
	return "3:04:05 PM" // medium
}

// localeFor maps a language.Tag to one of monday's Locale constants.
// monday accepts any monday.Locale string; we map the major locales we
// test against. Unknown tags fall back to en_US.
func localeFor(tag language.Tag) monday.Locale {
	base, _ := tag.Base()
	region, _ := tag.Region()
	composite := base.String() + "_" + region.String()
	switch composite {
	case "fr_FR":
		return monday.LocaleFrFR
	case "de_DE":
		return monday.LocaleDeDE
	case "ja_JP":
		return monday.LocaleJaJP
	case "es_ES":
		return monday.LocaleEsES
	case "en_GB":
		return monday.LocaleEnGB
	}
	// Fall back by base language alone.
	switch base.String() {
	case "fr":
		return monday.LocaleFrFR
	case "de":
		return monday.LocaleDeDE
	case "ja":
		return monday.LocaleJaJP
	case "es":
		return monday.LocaleEsES
	}
	return monday.LocaleEnUS
}
```

```go
// AssertDiagnostics matches directives against the diagnostic stream.
// Phase "lint" filters to warnings; all other phases filter to errors.
func AssertDiagnostics(t testing.TB, diags []ir.Diagnostic, dirs []ErrorDirective, phase string) {
	t.Helper()
	targetSeverity := ir.Error
	if phase == "lint" {
		targetSeverity = ir.Warning
	}
	var actual []string
	for _, d := range diags {
		if d.Severity == targetSeverity {
			actual = append(actual, d.Msg)
		}
	}
	relevant := Filter(dirs, phase)
	for _, e := range relevant {
		found := false
		for _, msg := range actual {
			if strings.Contains(msg, e.Substring) {
				found = true
				break
			}
		}
		if !found {
			t.Errorf("expected %s diagnostic containing %q, got: %v", phase, e.Substring, actual)
		}
	}
}
```

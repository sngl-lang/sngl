================================

func spin(n int) int {
    for {
        if n > 3 {
            break
        }
        n = n + 1
        continue
    }
    return n
}

func countTo(n int) string {
    var i = 0
    for i < n {
        i = i + 1
    } else {
        return "none"
    }
    return "some"
